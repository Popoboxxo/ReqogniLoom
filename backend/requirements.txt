# ReqFlow Backend Dependencies
# ARCH-L1: All backend systems

# Core Framework
# Django 5.2 LTS. 4.2 LTS reached end of life, so upstream no longer backports
# security fixes to the 4.2.x series — every newly disclosed Django CVE stays
# permanently unfixed there and the CI `pip-audit --desc` gate fails forever.
# The 4.2 pin was already blocking on six advisories at once:
#   CVE-2026-48587, CVE-2026-6873,  CVE-2026-8404  (fixed in 5.2.15)
#   CVE-2026-48588, CVE-2026-53877, CVE-2026-53878 (fixed in 5.2.16)
# 5.2 is the next LTS in Django's 2-year cadence (2.2 → 3.2 → 4.2 → 5.2) and is
# supported until April 2028. Floor set to the newest 5.2.x patch so all six
# advisories are covered; capped below 5.3 to stay on the LTS series.
# Requires Python >=3.10 (backend/Dockerfile runs 3.12) and PostgreSQL >=14
# (docker-compose runs pgvector/pgvector:pg16).
Django>=5.2.17,<5.3
# 3.17.2+ declares Django 5.2 support; the existing range already resolves
# there, so no floor change is required.
djangorestframework>=3.15,<4.0
drf-spectacular>=0.30.0,<1.0

# Database
psycopg2-binary>=2.9.12,<3.0

# Vector similarity search (REQ-L2-VS-004) — pgvector Django integration
# (VectorField, HnswIndex, CosineDistance). Requires the pgvector Postgres
# extension in the DB image (pgvector/pgvector:pg16, provisioned by devops).
pgvector>=0.5.0,<0.6.0

# Async task queue (ARCH-L1-016 ResilienceOrchestrator)
celery>=5.6.3,<6.0
django-celery-beat>=2.9.0,<3.0
redis>=5.0,<6.0

# Testing
# pytest floor raised to 9.0.3 for CVE-2025-71176 / PYSEC-2026-1845 (CWE-379):
# predictable /tmp/pytest-of-{user} basetemp directories on UNIX allow a local
# user to cause a DoS or possibly escalate privileges. Fixed only in 9.0.3 —
# there is no 8.x backport, so the CI pip-audit gate forces the major bump.
pytest>=9.0.3,<10.0
# pytest-django 4.12.0 is the first release whose own test matrix runs against
# pytest>=9; earlier 4.x releases predate pytest 9 entirely. Its install
# requirement is only `pytest>=7`, so the floor has to be pinned here.
# The former <4.12.1 cap existed only because pytest-django 4.12.1+ drops
# Django 4.2 support, and 4.13.0 against Django 4.2 blew up every
# @pytest.mark.django_db test with `AttributeError: type object
# 'PytestDjangoTestCase' has no attribute '_pre_setup_ran_eagerly'` — an
# internal hook that only exists on pytest-django builds targeting
# Django>=5.0. That cap was explicitly documented as a stopgap to be removed
# in the same change that upgrades Django; the Django 5.2 bump above is that
# change, so the cap is gone and the floor is raised to the first release that
# declares Django 5.2 support.
pytest-django>=4.13.0,<5.0
pypdf>=6.14.2,<7.0

# PDF generation (REQ-L2-AS-016, REQ-L1-023)
reportlab>=4.0,<5.0

uvicorn[standard]>=0.30,<1.0
